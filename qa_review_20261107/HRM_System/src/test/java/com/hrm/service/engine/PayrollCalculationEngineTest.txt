package com.hrm.service.engine;

import com.hrm.dto.AttendanceSummaryDTO;
import com.hrm.dto.EmployeeSalaryProfileDTO;
import com.hrm.dto.EmployeeSalaryProfileDTO.AllowanceItemDTO;
import com.hrm.model.PayrollLine;
import com.hrm.model.SalaryComponent;
import com.hrm.repository.SalaryComponentRepository;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.when;

class PayrollCalculationEngineTest {

    private SalaryComponentRepository componentRepository;
    private PayrollCalculationEngine engine;

    @BeforeEach
    void setUp() {
        componentRepository = Mockito.mock(SalaryComponentRepository.class);
        List<SalaryComponent> components = new ArrayList<>();
        components.add(new SalaryComponent("LUONG_CONG", "Lương ngày công", null, true));
        components.add(new SalaryComponent("TANG_CA", "Lương làm thêm giờ (OT)", null, true));
        components.add(new SalaryComponent("PHU_CAP", "Phụ cấp", null, true));
        components.add(new SalaryComponent("KHAU_TRU", "Khấu trừ", null, true));
        when(componentRepository.findAll()).thenReturn(components);

        engine = new PayrollCalculationEngine(componentRepository);
    }

    @Test
    @DisplayName("Kịch bản 1: Đối soát chuẩn nhân viên đủ công, có OT 4.5h, phụ cấp và chịu thuế TNCN")
    void testStandardCalculationWithTaxAndOvertime() {
        EmployeeSalaryProfileDTO profile = new EmployeeSalaryProfileDTO();
        profile.setEmployeeId(1L);
        profile.setBaseSalary(new BigDecimal("15000000"));
        List<AllowanceItemDTO> allowances = new ArrayList<>();
        allowances.add(new AllowanceItemDTO("AN_TRUA", "Phụ cấp ăn trưa", new BigDecimal("730000"), "THEO_KY"));
        allowances.add(new AllowanceItemDTO("XANG_XE", "Phụ cấp đi lại", new BigDecimal("500000"), "THEO_KY"));
        profile.setAllowances(allowances);

        AttendanceSummaryDTO attendance = new AttendanceSummaryDTO(
                1L,
                new BigDecimal("22"),
                new BigDecimal("21"),
                new BigDecimal("4.5"),
                new BigDecimal("1"),
                new BigDecimal("0")
        );

        List<PayrollLine> lines = engine.calculatePayrollLines(100L, profile, attendance);

        BigDecimal gross = BigDecimal.ZERO;
        BigDecimal deduction = BigDecimal.ZERO;

        for (PayrollLine line : lines) {
            if ("DEDUCT".equalsIgnoreCase(line.getDirectionSnapshot())) {
                deduction = deduction.add(line.getAmount());
            } else {
                gross = gross.add(line.getAmount());
            }
        }

        BigDecimal net = gross.subtract(deduction);

        // Khớp 100% với kết quả hệ thống thực tế đã chạy:
        assertEquals(new BigDecimal("16805284"), gross, "Tổng thu nhập Gross phải là 16.805.284 VNĐ");
        assertEquals(new BigDecimal("1750014"), deduction, "Tổng khấu trừ (BH + Thuế) phải là 1.750.014 VNĐ");
        assertEquals(new BigDecimal("15055270"), net, "Lương thực nhận Net phải là 15.055.270 VNĐ");
    }

    @Test
    @DisplayName("Kịch bản 2: Nhân viên thu nhập dưới 11 triệu không phải nộp thuế TNCN")
    void testLowSalaryNoTax() {
        EmployeeSalaryProfileDTO profile = new EmployeeSalaryProfileDTO();
        profile.setEmployeeId(2L);
        profile.setBaseSalary(new BigDecimal("8000000")); // Lương cơ bản 8 triệu
        profile.setAllowances(new ArrayList<>());

        AttendanceSummaryDTO attendance = new AttendanceSummaryDTO(
                2L,
                new BigDecimal("22"),
                new BigDecimal("22"),
                BigDecimal.ZERO,
                BigDecimal.ZERO,
                BigDecimal.ZERO
        );

        List<PayrollLine> lines = engine.calculatePayrollLines(101L, profile, attendance);

        BigDecimal gross = BigDecimal.ZERO;
        BigDecimal deduction = BigDecimal.ZERO;

        for (PayrollLine line : lines) {
            if ("DEDUCT".equalsIgnoreCase(line.getDirectionSnapshot())) {
                deduction = deduction.add(line.getAmount());
            } else {
                gross = gross.add(line.getAmount());
            }
        }

        // Lương 8tr -> Bảo hiểm 10.5% = 840.000đ, Thuế = 0 (vì sau giảm trừ bản thân 11tr thu nhập tính thuế âm)
        assertEquals(new BigDecimal("8000000"), gross);
        assertEquals(new BigDecimal("840000"), deduction);
        assertEquals(new BigDecimal("7160000"), gross.subtract(deduction));
    }
}
